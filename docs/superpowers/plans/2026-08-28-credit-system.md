# Credit System Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Meter every LLM message, token, tool call, and skill run against a tenant-scoped credit balance, replacing the two monthly quota functions that enforce today.

**Architecture:** A new foundation package (`packages/foundation/credits`) owns three tables - `credit_accounts` (materialized balance), `credit_grants` (dated buckets), `credit_ledger` (audit rows) - and one Postgres function, `spend_credits()`, which is the only writer of `balance_micro`. Prices live in `credit_rates`, versioned and ops-editable. The API/worker call the function through Drizzle; the agent orchestrator calls it through its existing raw `pg` pool.

**Tech Stack:** TypeScript, Drizzle ORM, Postgres (Supabase, transaction pooler), Hono on Lambda, Mastra orchestrator on a GCP VM, vitest, Next.js + shadcn/ui.

**Spec:** `docs/superpowers/specs/2026-08-28-credit-system-design.md` - read it before Task 1. Every design rationale lives there; this plan does not repeat it.

## Global Constraints

- **1 credit = 1,000,000 micro.** All amounts are `bigint` micro. No floats anywhere. Round up, once, in `rate.ts`.
- **1 credit = 1 US cent** at the seeded rates (spec open question 1). Rates carry no markup.
- **Nothing outside `spend_credits()` writes `credit_accounts.balance_micro`.**
- **Invariant:** `balance_micro` always equals the sum of `(amount_micro - spent_micro)` over live, unexpired grants. Every task that touches money must preserve it.
- **`job_id` carries no foreign key.** `agent_tasks` is product schema (`products/agent-platform/packages/schema/agents.ts`); a foundation table must not reference it.
- **Permissions are `(resource, action)` pairs** with `action` in `create|read|update|delete`. `credits:spend` is not expressible - use `('credits','create')`.
- **Package scope is `@serverless-saas/*`.** AWS resources use the `project-context` prefix. Never mix them.
- **Drizzle `mode: 'bigint'` returns a JS `BigInt`, which `JSON.stringify` throws on.** Serialize every micro amount as a string at the route boundary.
- **The credit pre-check fails CLOSED on a database error.** The quota functions it replaces failed open; this is a deliberate change (spec open question 6).
- **Deploy split:** `./deploy.sh` restarts only `web-frontend` and `api` on the VM. Orchestrator changes need `pm2 restart agent-orchestrator`. Lambda changes need `sam deploy` with foundation and product packages rebuilt first.

## Test database

Use the local dev database from `infra/vm` (or `docker run -e POSTGRES_PASSWORD=postgres -p 5433:5432 pgvector/pgvector:pg16`), exported as `TEST_DATABASE_URL`.   updatedAt: timestamp('updated_at', { withTimezone: true }).notNull().defaultNow(),

```ts
  id: uuid('id').primaryKey().defaultRandom(),

  id: uuid('id').primaryKey().defaultRandom(),

  createdAt: timestamp('created_at', { withTimezone: true }).notNull().defaultNow(),

}, (t) => [
```

```ts
// packages/foundation/credits/src/rate.ts
export const MICRO_PER_CREDIT = 1_000_000n;

export type PricingSchema =
  | { per_million_tokens_micro: { input: number; output: number } }
  | { per_message_micro: number }
  | { per_call_micro: number }
  | { per_run_micro: number };

export interface Usage {
  inputTokens?: number;
  outputTokens?: number;
  count?: number;
}

/** Cost in micro-credits. Rounds UP exactly once, after summing components. */
export function costMicro(schema: PricingSchema, usage: Usage): bigint {
  if ('per_million_tokens_micro' in schema) {
    const { input, output } = schema.per_million_tokens_micro;
    const numerator =
      BigInt(usage.inputTokens ?? 0) * BigInt(input) +
      BigInt(usage.outputTokens ?? 0) * BigInt(output);
    return ceilDiv(numerator, 1_000_000n);
  }
  const flat =
    'per_message_micro' in schema ? schema.per_message_micro
    : 'per_call_micro' in schema ? schema.per_call_micro
    : schema.per_run_micro;
  return BigInt(flat) * BigInt(usage.count ?? 0);
}

function ceilDiv(n: bigint, d: bigint): bigint {
  return n <= 0n ? 0n : (n + d - 1n) / d;
}
```

```ts
// packages/foundation/credits/src/read.ts
import { and, desc, eq, gt, isNull, or, sql } from 'drizzle-orm';
import { db, creditAccounts, creditGrants, creditLedger, creditRates } from '@serverless-saas/database';
import type { PricingSchema } from './rate';

export interface GrantSummary {
  grantType: string;
  amountMicro: bigint;
  spentMicro: bigint;
  expiresAt: Date | null;
}

export async function getBalance(tenantId: string) {
  const [account] = await db.select().from(creditAccounts)
    .where(eq(creditAccounts.tenantId, tenantId)).limit(1);
  const grants = await db.select().from(creditGrants).where(and(
    eq(creditGrants.tenantId, tenantId),
    or(isNull(creditGrants.expiresAt), gt(creditGrants.expiresAt, new Date())),
    sql`${creditGrants.spentMicro} < ${creditGrants.amountMicro}`,
  ));
  return {
    balanceMicro: account?.balanceMicro ?? 0n,
    unlimited: await isUnlimited(tenantId),
    grants: grants.map<GrantSummary>(g => ({
      grantType: g.grantType, amountMicro: g.amountMicro,
      spentMicro: g.spentMicro, expiresAt: g.expiresAt,
    })),
  };
}

export async function getLedger(tenantId: string, opts: { limit?: number; before?: Date; kind?: string }) {
  const conditions = [eq(creditLedger.tenantId, tenantId)];
  if (opts.before) conditions.push(sql`${creditLedger.createdAt} < ${opts.before}`);
  if (opts.kind) conditions.push(eq(creditLedger.kind, opts.kind as never));
  return db.select().from(creditLedger).where(and(...conditions))
    .orderBy(desc(creditLedger.createdAt)).limit(Math.min(opts.limit ?? 50, 200));
}

/** Active rate for (resourceType, subject), falling back to the '*' subject row. */
export async function resolveRate(resourceType: string, subject: string) {
  const rows = await db.select().from(creditRates).where(and(
    eq(creditRates.resourceType, resourceType as never),
    eq(creditRates.isActive, true),
    sql`${creditRates.subject} in (${subject}, '*')`,
  ));
  // exact subject wins over the wildcard
  const row = rows.find(r => r.subject === subject) ?? rows.find(r => r.subject === '*');
  return row ? { id: row.id, version: row.version, schema: row.pricingSchema as PricingSchema } : null;
}
```

```ts
export async function isUnlimited(tenantId: string): Promise<boolean> {
  const [row] = await db.execute(sql`
    select coalesce(tfo.unlimited, pe.unlimited, false) as unlimited
      from subscriptions s
      join features f on f.key = 'credits'
      left join plan_entitlements pe on pe.plan = s.plan and pe.feature_id = f.id
      left join tenant_feature_overrides tfo
             on tfo.tenant_id = s.tenant_id and tfo.feature_id = f.id
            and tfo.revoked_at is null and tfo.deleted_at is null
            and (tfo.expires_at is null or tfo.expires_at > now())
     where s.tenant_id = ${tenantId} and s.status in ('active','trialing')
     limit 1
  `) as unknown as { unlimited: boolean }[];
```

```ts
import postgres from 'postgres';
import { grantCredits, isUnlimited, MICRO_PER_CREDIT } from '@serverless-saas/credits';

const dryRun = process.argv.includes('--dry-run');
const sql = postgres(process.env.DATABASE_URL!, { max: 1 });

function endOfMonth(): Date {
  const now = new Date();
  return new Date(Date.UTC(now.getUTCFullYear(), now.getUTCMonth() + 1, 1));
}

async function run() {
  const tenants = await sql<{ tenant_id: string; value_limit: number | null }[]>`
    select s.tenant_id, coalesce(tfo.value_limit, pe.value_limit) as value_limit
      from subscriptions s
      join features f on f.key = 'credits'
      left join plan_entitlements pe on pe.plan = s.plan and pe.feature_id = f.id
      left join tenant_feature_overrides tfo
             on tfo.tenant_id = s.tenant_id and tfo.feature_id = f.id
            and tfo.revoked_at is null and tfo.deleted_at is null
     where s.status in ('active', 'trialing')
  `;

  let granted = 0;
  for (const t of tenants) {
    if (await isUnlimited(t.tenant_id)) {
      console.log(`skip ${t.tenant_id}: unlimited`);
      continue;
    }
    if (t.value_limit == null) {
      console.warn(`skip ${t.tenant_id}: no credits entitlement - check the plan seed`);
      continue;
    }
    const [existing] = await sql`
      select 1 from credit_ledger where idempotency_key = ${'backfill:' + t.tenant_id} limit 1`;
    if (existing) {
      console.log(`skip ${t.tenant_id}: already backfilled`);
      continue;
    }
    const amountMicro = BigInt(t.value_limit) * MICRO_PER_CREDIT;
    console.log(`${dryRun ? 'would grant' : 'grant'} ${t.value_limit} credits to ${t.tenant_id}`);
    if (!dryRun) {
      await sql`insert into credit_accounts (tenant_id) values (${t.tenant_id})
                on conflict do nothing`;
      await grantCredits({
        tenantId: t.tenant_id,
        amountMicro,
        key: `backfill:${t.tenant_id}`,
        grantType: 'subscription',
        expiresAt: endOfMonth(),
        reason: 'opening balance at credit system rollout',
      });
    }
    granted += 1;
  }
  console.log(`${dryRun ? 'would grant' : 'granted'} ${granted} of ${tenants.length} tenants`);
  await sql.end();
}

run().catch(async (err) => { console.error('backfill failed', err); await sql.end(); process.exit(1); });
```

```ts
// apps/agent-orchestrator/src/credits.ts (additions)
import { costMicro } from '@serverless-saas/credits';

/** Tokens assumed per workflow step when no better estimate exists. Tune from token_costs. */
const AVERAGE_STEP_TOKENS = { input: 4000, output: 1000 };

export async function estimateTaskMicro(stepCount: number, model: string): Promise<bigint> {
  const rate = await resolveRateCached('llm_tokens', model);
  if (!rate) return 0n;
  return costMicro(rate.schema, {
    inputTokens: AVERAGE_STEP_TOKENS.input * stepCount,
    outputTokens: AVERAGE_STEP_TOKENS.output * stepCount,
  });
}

/** Positive credits the tenant back; negative debits the shortfall; 0n means do nothing. */
export function settleAmount(args: { estimateMicro: bigint; actualMicro: bigint }): bigint {
  return args.estimateMicro - args.actualMicro;
}

export async function settleTask(args: {
  tenantId: string; taskId: string; agentId: string;
  model: string; inputTokens: number; outputTokens: number; estimateMicro: bigint;
}): Promise<void> {
  const rate = await resolveRateCached('llm_tokens', args.model);
  if (!rate) return;
  const actualMicro = costMicro(rate.schema, {
    inputTokens: args.inputTokens, outputTokens: args.outputTokens,
  });
  const delta = settleAmount({ estimateMicro: args.estimateMicro, actualMicro });
  if (delta === 0n) return;
  await callSpend({
    tenantId: args.tenantId, amountMicro: delta, key: `task:${args.taskId}:settle`,
    kind: 'settle', actorId: args.agentId, actorType: 'agent',
    rateId: rate.id, rateVersion: rate.version,
    jobId: args.taskId, jobType: 'agent_task',
    grantType: delta > 0n ? 'refund' : null,
    reason: 'settle estimate to actual',
  });
}

export async function refundTask(args: { tenantId: string; taskId: string }): Promise<void> {
  const pool = getPool();
  const res = await pool.query<{ total: string }>(
    `select coalesce(sum(amount_micro), 0) as total from credit_ledger
      where tenant_id = $1 and idempotency_key in ($2, $3)`,
    [args.tenantId, `task:${args.taskId}`, `task:${args.taskId}:settle`],
  );
  const net = BigInt(res.rows[0]?.total ?? '0');   // negative if the tenant is down money
  if (net >= 0n) return;                            // nothing was charged, or already refunded
  await callSpend({
    tenantId: args.tenantId, amountMicro: -net, key: `task:${args.taskId}:refund`,
    kind: 'refund', grantType: 'refund', jobId: args.taskId, jobType: 'agent_task',
    reason: 'task failed',
  });
}
```
